algoritmo
    declare Alunos[20] registro (
        nome literal,
        nota_bimestre[4] numérico,
    )

    declare i, j, notas_da_turma, media, OP, media_soma_por_aluno, media_do_aluno numérico
    escreva " 1_ Cadastrar"
    escreva " 2_ Exibir os alunos e notas cadastrados"
    escreva " 3_ Exibir a média de cada aluno"
    escreva " 4_ Exibir a média da classe"
    escreva " 5_ Exibir quais alunos ficaram acima da média da turma"

    leia  OP
    
    caso OP início
        igual a 1 então início
            para i = 1 até 20 faça passo 1 início
                leia Alunos.nome[i]
                para j = 1 até 4 faça passo 1
                    leia Alunos.nota_bimestre[j]
                    media_turma_soma = media_turma_soma + Alunos.nota_bimestre[j]
                fim           
            fim
        fim

        igual a 2 então início
            para i = 1 até 20 faça passo 1 início
                escreva "Aluno", Alunos.nome[i]     
                escreva " Notas: "
                para j = 1 até 4 faça passo 1 início
                    escreva Alunos.nota_bimestre[j]
                fim
            fim
        fim

        igual a 3 então início
            para i = 1 até 20 faça passo 1 início
                media_soma_por_aluno = 0
                media = 0
                escreva Aluno.nome[i]
                para j = 1 até 4 faça passo 1 início
                    media_soma_por_aluno = media_soma_por_aluno + Aluno.nota_bimestre[j]
                fim
                media_do_aluno = media_soma_por_aluno / 4
                escreva media
            fim
        fim

        igual a 4 então início
            media = media_turma_soma / 20
            escreva media
        fim

        igual a 5 então início
            para i = 1 até 20 faça passo 1 início
                para j = 1 até 4 faça passo 1 início
                    media_soma_por_aluno = media_soma_por_aluno + Aluno.nota_bimestre[j]
                fim
                media_do_aluno = media_soma_por_aluno / 4

                se media_do_aluno > media início
                    escreva Aluno.nome[i]
                fim
            escreva "\n"
            fim
        fim
fim algoritmo
