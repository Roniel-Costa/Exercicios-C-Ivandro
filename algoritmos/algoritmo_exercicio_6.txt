Função numerico converte_totalsegundos ( horas, minutos, segundos numérico)
	Converte_totalsegundos <- (horas * 3600) + (minutos * 60) + segundos
fim função

função numérico converte_horas (total_segundos numerico)
	converte_horas <- QUOCIENTE(total_segundos,3600)
fim função

função numérico converte_minutos(total_segundos numérico)
	declare sobra numérico
	sobra <- RESTO(total_segundos, 3600)
	converte_minutos <- QUOCIENTE(sobra, 60)
fim função

função numérico converte_segundos (total_segundos numérico)
	declare sobra numérico
	sobra <-  RESTO (total_segundos, 3600)
	converte_segundos <- RESTO (sobra, 60)
fim função
